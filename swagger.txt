const setupSwagger = (app) => {
  // Save standalone openapi.json and openapi.yaml to disk
  try {
    const jsonPath = path.resolve(__dirname, '../../openapi.json');
    const yamlPath = path.resolve(__dirname, '../../openapi.yaml');
    fs.writeFileSync(jsonPath, JSON.stringify(openApiDocumentation, null, 2));
    fs.writeFileSync(yamlPath, yaml.dump(openApiDocumentation));
  } catch (err) {
    // Non-fatal if read-only filesystem
  }

  // Interactive Swagger UI
  app.use('/api-docs', swaggerUi.serve, swaggerUi.setup(openApiDocumentation, {
    customSiteTitle: 'OneLife API Documentation',
    customCss: '.swagger-ui .topbar { display: block; }'
  }));

  // Direct OpenAPI JSON specification endpoint
  app.get('/api-docs.json', (req, res) => {
    res.setHeader('Content-Type', 'application/json');
    res.send(openApiDocumentation);
  });

  // Direct OpenAPI YAML specification endpoint
  app.get('/api-docs.yaml', (req, res) => {
    res.setHeader('Content-Type', 'text/yaml');
    res.send(yaml.dump(openApiDocumentation));
  });
};

module.exports = {
  openApiDocumentation,
  setupSwagger
};
